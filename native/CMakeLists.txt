cmake_minimum_required(VERSION 3.25)
project(v8_chroma_native LANGUAGES CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The drawing backend currently supports Windows x64 only. Portable Rust core checks do not require this backend.")
endif()
if(MSVC AND NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
    message(FATAL_ERROR "The Windows drawing backend has not been ported to ARM64.")
endif()
if(NOT MSVC AND CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]|aarch64")
    message(FATAL_ERROR "The Windows drawing backend has not been ported to ARM64.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WINVER=0x0a00 _WIN32_WINNT=0x0a00)
if(MSVC)
    add_compile_options(/W4 /utf-8 /permissive-)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
    add_compile_options(-Wall -Wextra)
    # MinGW is an optional developer toolchain; keep its runtime inside binaries.
    if(NOT V8_CHROMA_ZIG_TOOLCHAIN)
        add_link_options(-static -static-libgcc -static-libstdc++)
    endif()
endif()

function(output target name group)
    set_target_properties(${target} PROPERTIES
        OUTPUT_NAME "${name}" PREFIX ""
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${group}/$<0:>"
        ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/${group}/$<0:>")
endfunction()

function(console target)
    if(MINGW)
        target_link_options(${target} PRIVATE -municode)
    endif()
endfunction()

add_library(theme_engine SHARED src/probe.cpp src/window_theme.cpp src/raster_theme.cpp src/editor_selection.cpp)
target_link_libraries(theme_engine PRIVATE gdi32 user32 dwmapi)
output(theme_engine ThemeProbe "")
install(TARGETS theme_engine RUNTIME DESTINATION assets)
install(TARGETS theme_engine RUNTIME DESTINATION native/bin)

option(V8_CHROMA_BUILD_ICON_PROTOTYPE "Build the opt-in, separate toolbar icon experiment" OFF)
if(V8_CHROMA_BUILD_ICON_PROTOTYPE)
    add_library(toolbar_icons SHARED prototype/toolbar_icons.cpp)
    output(toolbar_icons ToolbarIcons prototype)
endif()

option(V8_CHROMA_BUILD_FIXTURES "Build owned DLLs and process fixtures for tests" ON)
if(V8_CHROMA_BUILD_FIXTURES)
    add_library(mock_cairo SHARED tests/mock_cairo.cpp)
    output(mock_cairo cairo demo)
    if(V8_CHROMA_BUILD_ICON_PROTOTYPE)
        add_library(icon_test_graphics SHARED prototype/test_graphics.cpp)
        target_link_libraries(icon_test_graphics PRIVATE mock_cairo)
        output(icon_test_graphics grphcs prototype-fixtures)
        add_executable(toolbar_icons_test prototype/test_host.cpp)
        console(toolbar_icons_test)
        output(toolbar_icons_test toolbar_icons_test prototype-fixtures)
    endif()
    add_library(mock_frame SHARED tests/frame.cpp)
    target_link_libraries(mock_frame PRIVATE gdi32 user32)
    output(mock_frame frame demo)
    add_library(mock_graphics SHARED tests/grphcs.cpp)
    target_link_libraries(mock_graphics PRIVATE mock_cairo)
    output(mock_graphics grphcs demo)
    add_executable(probe_target tests/target.cpp)
    target_link_libraries(probe_target PRIVATE mock_frame mock_graphics gdi32 user32 dwmapi)
    console(probe_target)
    output(probe_target probe_target demo)
    add_executable(raster_test tests/raster.cpp src/raster_theme.cpp)
    console(raster_test)
    output(raster_test raster_test demo)
    add_executable(editor_selection_test tests/editor_selection.cpp src/editor_selection.cpp)
    console(editor_selection_test)
    output(editor_selection_test editor_selection_test demo)
    add_executable(mode_host tests/mode_host.cpp)
    console(mode_host)
    output(mode_host mode_host demo)
    install(TARGETS mock_cairo mock_frame mock_graphics probe_target raster_test editor_selection_test mode_host
        RUNTIME DESTINATION tests/fixtures)
    install(TARGETS mock_cairo mock_frame mock_graphics probe_target raster_test editor_selection_test mode_host
        RUNTIME DESTINATION native/bin/demo)

    # Link owned graphics fixtures against the mock import library, then load
    # the real, user-installed Cairo at test time. No 1C binary is distributed.
    add_library(real_frame SHARED tests/frame.cpp)
    target_link_libraries(real_frame PRIVATE gdi32 user32)
    output(real_frame frame real)
    add_library(real_graphics SHARED tests/grphcs_real.cpp)
    target_link_libraries(real_graphics PRIVATE mock_cairo)
    output(real_graphics grphcs real)
    add_library(real_browser SHARED tests/browser_real.cpp)
    target_link_libraries(real_browser PRIVATE mock_cairo user32)
    output(real_browser WebKit real)
    add_executable(real_selection_host tests/real_selection_host.cpp)
    target_link_libraries(real_selection_host PRIVATE user32)
    console(real_selection_host)
    output(real_selection_host real_selection_host real)
    install(TARGETS real_frame real_graphics real_browser real_selection_host RUNTIME DESTINATION native/bin/real)
endif()
