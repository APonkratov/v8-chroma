cmake_minimum_required(VERSION 3.25)
project(v8_chroma_native LANGUAGES CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The drawing backend currently supports Windows x64 only. Portable Rust core checks do not require this backend.")
endif()
if(MSVC AND NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
    message(FATAL_ERROR "The Windows drawing backend has not been ported to ARM64.")
endif()
if(NOT MSVC AND CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]|aarch64")
    message(FATAL_ERROR "The Windows drawing backend has not been ported to ARM64.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WINVER=0x0a00 _WIN32_WINNT=0x0a00)
if(MSVC)
    add_compile_options(/W4 /utf-8 /permissive-)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
    add_compile_options(-Wall -Wextra)
    # MinGW is an optional developer toolchain; keep its runtime inside binaries.
    if(NOT V8_CHROMA_ZIG_TOOLCHAIN)
        add_link_options(-static -static-libgcc -static-libstdc++)
    endif()
endif()

function(output target name group)
    set_target_properties(${target} PROPERTIES
        OUTPUT_NAME "${name}" PREFIX ""
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/${group}/$<0:>"
        ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/${group}/$<0:>")
endfunction()

function(console target)
    if(MINGW)
        target_link_options(${target} PRIVATE -municode)
    endif()
endfunction()

target_link_libraries(theme_engine PRIVATE gdi32 user32 dwmapi version)
output(theme_engine ThemeProbe "")
install(TARGETS theme_engine RUNTIME DESTINATION assets)
install(TARGETS theme_engine RUNTIME DESTINATION native/bin)
